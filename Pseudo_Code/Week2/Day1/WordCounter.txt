CLASS WordCounter
    METHOD countWords(text)
        words = split text into tokens by space
        CREATE Map<String, Integer> wordCount
        FOR each word in words
            word = lowercase(word)
            word = removePunctuation(word)
            INCREMENT wordCount[word] by 1
        SORT wordCount by frequency (descending)
        PRINT wordCount

MAIN
    text = "The news today is about the rise in technology trends."
    WordCounter.countWords(text)
