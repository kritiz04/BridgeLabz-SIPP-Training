CLASS TaskOrganizer
    ATTRIBUTES: LinkedHashMap<Integer, String> tasks
    counter = 1

    METHOD addTask(description)
        tasks[counter] = description
        counter++

    METHOD removeTask(taskId)
        DELETE tasks[taskId]

    METHOD showTasks()
        FOR each (id, task) in tasks
            PRINT id + " → " + task

MAIN
    organizer = TaskOrganizer()
    organizer.addTask("Check emails")
    organizer.addTask("Prepare sales report")
    organizer.addTask("Call client")
    organizer.removeTask(2)
    organizer.showTasks()
