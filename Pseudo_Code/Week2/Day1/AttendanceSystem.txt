CLASS AttendanceSystem
    ATTRIBUTES: attendance[students][days]

    METHOD markAttendance(studentId, day, value)
        attendance[studentId][day] = value

    METHOD updateAttendance(studentId, day, value)
        attendance[studentId][day] = value

    METHOD generateReport(studentId)
        count = 0
        FOR each day
            IF attendance[studentId][day] == 'P'
                count++
        PRINT "Total Present = " + count

MAIN
    system = AttendanceSystem(30 students, 31 days)
    system.markAttendance(1, 1, 'P')
    system.markAttendance(1, 2, 'A')
    system.updateAttendance(1, 2, 'P')
    system.generateReport(1)
