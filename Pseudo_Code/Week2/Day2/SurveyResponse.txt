CLASS SurveyResponse
    ATTRIBUTES: restaurantName, rating

CLASS SurveyAnalyzer
    METHOD generateReport(responses)
        CREATE Map<String, List<Integer>> groupedRatings
        FOR response in responses
            ADD response.rating to groupedRatings[response.restaurantName]

        CREATE Map<String, Double> avgRatings
        FOR each restaurant in groupedRatings
            avg = SUM(ratings)/COUNT(ratings)
            avgRatings[restaurant] = avg

        PRINT avgRatings
        FIND restaurant with MAX avg
        PRINT "Best Restaurant = " + restaurant
        FIND restaurant with MIN avg
        PRINT "Needs improvement = " + restaurant

MAIN
    responses = [
        SurveyResponse("PizzaHut", 4),
        SurveyResponse("PizzaHut", 5),
        SurveyResponse("KFC", 3),
        SurveyResponse("KFC", 4),
        SurveyResponse("Subway", 5),
        SurveyResponse("Subway", 4),
        SurveyResponse("Subway", 5)
    ]
    analyzer = SurveyAnalyzer()
    analyzer.generateReport(responses)
    // Output: PizzaHut=4.5, KFC=3.5, Subway=4.67
    // Best=Subway, Needs Improvement=KFC
