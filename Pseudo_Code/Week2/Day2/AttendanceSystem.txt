CLASS AttendanceSystem
    ATTRIBUTES: attendance[30][30]  // 30 students, 30 days

    METHOD markAttendance(studentId, day, value)  // 1 = Present, 0 = Absent
        attendance[studentId][day] = value

    METHOD dailySummary(day)
        count = 0
        FOR student=1 TO 30
            IF attendance[student][day] == 1
                count++
        PRINT "Present count on Day " + day + " = " + count

    METHOD monthlySummary(studentId)
        total = 0
        FOR day=1 TO 30
            total += attendance[studentId][day]
        PRINT "Student " + studentId + " attended " + total + " days"

    METHOD defaulterList()
        FOR student=1 TO 30
            total = SUM of attendance[student][*]
            IF total < 22  // < 75% of 30
                PRINT "Defaulter → Student " + student

MAIN
    system = AttendanceSystem()
    system.markAttendance(1, 1, 1)  // Student 1 present on Day 1
    system.markAttendance(3, 1, 0)  // Student 3 absent on Day 1
    system.dailySummary(10)
    system.monthlySummary(5)
    system.defaulterList()
